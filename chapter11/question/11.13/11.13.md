# プログラミング課題11.13
## 課題内容
**より安全なサーバ** サーバでpopenを使うのは、非常に危険である。この危険に対処する方法は２種類ある。
１つは、より柔軟だがより安全なsanitize関数を書くというものである。例えば、ピリオド、ハイフン、スペースなどが含まれているディレクトリ名には問題はない。
また、これらの文字に特別な意味を与えているのは、シェルに過ぎない。
より便利で安全な文字列クリーニング関数を書きなさい。
もう１つの方法は、popenを使わず、fork,exec,dupなどなどを使うというものである。
この方法を使ってrlsd.cを書き換えなさい。waitを使う必要はあるか。その理由はどうか、説明しなさい。

#include <ctype.h>
#include <limits.h>
#include <netdb.h>
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <strings.h>
#include <sys/socket.h>
#include <sys/types.h>
#include <unistd.h>

#define PORTNUM 15000
#define HOSTLEN 256
#define oops(msg)                                                              \
  {                                                                            \
    perror(msg);                                                               \
    exit(1);                                                                   \
  }
void sanitize(char *str);

int main(int ac, char *av[]) {
  struct sockaddr_in saddr;
  struct hostent *hp;
  char hostname[HOSTLEN];
  int sock_id, sock_fd;
  FILE *sock_fpi, *sock_fpo;
  FILE *pipe_fp;
  char dirname[PATH_MAX];

  char command[BUFSIZ];
  int dirlen, c;

  /* ステップ1 : カーネルにソケットを要求する */

  sock_id = socket(PF_INET, SOCK_STREAM, 0);
  if (sock_id == -1) {
    oops("socket");
  }

  /** ステップ２：ソケットにアドレスをバインドする。アドレスはホスト、ポート **/

  memset((void *)&saddr, 0, sizeof(saddr));
  gethostname(hostname, HOSTLEN);
  hp = gethostbyname(hostname);
  memcpy((void *)&saddr.sin_addr, hp->h_addr, hp->h_length);
  saddr.sin_addr.s_addr = htonl(INADDR_ANY);
  saddr.sin_port = htons(PORTNUM);
  saddr.sin_family = AF_INET;
  if (bind(sock_id, (struct sockaddr *)&saddr, sizeof(saddr)) != 0)
    oops("bind");

  /** ステップ3 : キューサイズを１として、ソケットを着信できる状態にする。 */
  if (listen(sock_id, 1) != 0)
    oops("listen");
  /*
   *  メインループ: accept(), write(),close()
   */

  while (1) {
    sock_fd = accept(sock_id, NULL, NULL);
    if (sock_fd == -1)
      oops("accept");

    /* 読み出し方向をストリームとしてオープン*/
    if ((sock_fpi = fdopen(sock_fd, "r")) == NULL)
      oops("fdopen reading");

    if (fgets(dirname, sizeof(dirname), sock_fpi) == NULL)
      oops("reading dirname");
    sanitize(dirname);

    /* 書き込み方向をストリームとしてオープン*/
    if ((sock_fpo = fdopen(sock_fd, "w")) == NULL)
      oops("fdopen writing");

    snprintf(command, sizeof(command), "ls %s", dirname);
    if ((pipe_fp = popen(command, "r")) == NULL)
      oops("popen");

    while ((c = getc(pipe_fp)) != EOF)
      putc(c, sock_fpo);
    pclose(pipe_fp);
    fclose(sock_fpo);
    fclose(sock_fpi);
  }
}

void sanitize(char *str)
/*
 *  誰かが "; rm*"のようなディレクトリ名を送ってきたときに、素直に" ls ; rm *"
 * というコマンドを作るのでは非常にまずい。
 *
 *  そこでスラッシュと英数字以外の全ての文字を取り除く。
 *  もっと良い方法がある。練習問題を参照のこと。
 */
{
  char *src, *dest;

  for (src = dest = str; *src; src++)
    if (*src == '/' || isalnum(*src))
      *dest++ = *src;
  *dest = '\0';
}

```

問題の箇所を確認してみよう。
まずはsanitize関数だ。
これはstrという文字列をsrcというポインタで走査し、destにsrcを代入することで書き換えている。
destは/か英数字でなければポインタが++されないが、srcは++されるので、結果的に不要な文字は上から塗り潰される形になる。
srcを全て走査したところで*destの内容を'\0'にすれば、不要な文字は前詰めされた文字列に書き換えるという仕様になっている。

次に、popenを使わず、fork,exec,dupなどなどを使うと安全にできるとのやり方も記載されているが
これは、これはなぜだろうか？

## sanitize関数で安全にする方法について
### ピリオド、ハイフン、スペースなどが含まれているディレクトリは問題がない
ディレクトリ名にこれらを含んだ場合も正しく処理できるようにする。
おそらくだが、既存のif文にor条件を追加するだけで実現できるはずだ。

### ディレクトリ名はアスタリスクやセミコロンを含むことができる。
この２つはシェルが特別な意味があると認識している。
つまり、シェル上で意味を理解できればよいのだ。シェルはどの文字でこの２つの文字をエスケープできるのだろうか？

size_t sanitize(char *src, char *dest, size_t size)
/*

 */
{
  size_t used = 0;

  for (; *src; src++) {
    if (*src == '*' || *src == ';' || *src == ' ') {
      if (used + 2 >= size)
        break;
      dest[used++] = '\\';
      dest[used++] = *src;
    } else if (*src == '/' || isalnum(*src) || *src == '.' || *src == '-') {
      if (used + 1 >= size)
        break;
      dest[used++] = *src;
    }

  if (size > 0)
    dest[used] = '\0';

  return used;
}

## fork,exec,dup2で実装するのはどうすればいいか？
popenをfork,exec,dup2で実装するのは以前もfork,exec,dup2を使ってpopenを書いたときのように実装すればよい。

### コード
```

 */

#include <ctype.h>
#include <limits.h>
#include <netdb.h>

int main(int ac, char *av[]) {
  struct sockaddr_in saddr;
  struct hostent *hp;
  char hostname[HOSTLEN];

  FILE *sock_fpi, *sock_fpo;

  /* ステップ1 : カーネルにソケットを要求する */

  sock_id = socket(PF_INET, SOCK_STREAM, 0);

  saddr.sin_port = htons(PORTNUM);
  saddr.sin_family = AF_INET;
  if (bind(sock_id, (struct sockaddr *)&saddr, sizeof(saddr)) != 0)
    oops("bind");

  /** ステップ3 : キューサイズを１として、ソケットを着信できる状態にする。 */
  if (listen(sock_id, 1) != 0)
    oops("listen");
  /*
   *  メインループ: accept(), write(),close()

  while (1) {
    sock_fd = accept(sock_id, NULL, NULL);
    if (sock_fd == -1)
      oops("accept");

    /* 読み出し方向をストリームとしてオープン*/
    if ((sock_fpi = fdopen(sock_fd, "r")) == NULL)
      oops("fdopen reading");

    if (fgets(dirname, sizeof(dirname), sock_fpi) == NULL)
      oops("reading dirname");

    dirname[strcspn(dirname, "\r\n")] = '\0'; // 改行削除

    int pid = -1;

    int toChild[2], fromChild[2];
    if (pipe(toChild) == -1) {
      perror("pipe");
      continue;
    }
    if (pipe(fromChild) == -1) {
      perror("pipe");
      continue;
    }

    if ((pid = fork()) == -1) {
      perror("fork");
      fclose(sock_fpo);
      fclose(sock_fpi);
      continue;
    }

    if (pid == 0) {
      close(fromChild[0]); // 不要なfdを閉じる
      close(toChild[1]);   // 不要なfdを閉じる
      dup2(toChild[0], STDIN_FILENO);
      dup2(fromChild[1], STDOUT_FILENO);
      close(toChild[0]);
      close(fromChild[1]);

      char *argv[] = {"ls", dirname, NULL};
      execvp("ls", argv);
      perror("exec");
      exit(1);
    }

    close(toChild[0]);
    close(fromChild[1]);
    close(toChild[1]); // 不要なfdを閉じる

    FILE *fromChildFp = fdopen(fromChild[0], "r");
    char buf[BUFSIZ];

    /* 書き込み方向をストリームとしてオープン*/
    if ((sock_fpo = fdopen(sock_fd, "w")) == NULL)
      oops("fdopen writing");

    while ((fgets(buf, sizeof(buf), fromChildFp) != NULL)) {
      fputs(buf, sock_fpo);
    }

    fclose(fromChildFp);

    wait(NULL);

    fclose(sock_fpo);
    fclose(sock_fpi);
  }
}

## まとめ
sanitizeでpopenの危険性を完全になくすことはできない。
その後の仕様や設計変更に対しては守ることができないし、全てのパターンを網羅するのは漏れのリスクがどうしても出る。
そのため、サーバー、クライアント間の通信で、クライアント側にサーバーのPCのコマンドを使わせるのにpopen()を使うのは危険である。
サーバー、クライアント間の通信に関わらず、基本的にユーザーにpopenの第１引数のコマンドの一部を入力させるような設計は危険である。

fork(),exec()を使った外部コマンドの利用にすれば、クライアントから送られた文字列を確実にexecコマンドの引数にできるため、安全な設計になる。
ユーザーからの入力を受け取ってコマンドを実行するときは、fork,execを使って実装するようにしよう。


## 今回の学び
子プロセスでexecを行うときの引数を予め欲しい場合、
親プロセス側でコマンドの引数などを変数に格納しておいて、fork()することで、forkした先でも変数を引き継げる。
親プロセスから子プロセスへ送る情報がexecの引数程度で十分な場合、pipeによる通信でコマンドの引数などを渡さなくてもこの方法で渡すことができる。
execしたあとにやり取りする必要があるデータがある場合は、pipeを使おう。
